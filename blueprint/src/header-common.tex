% Real space characters in the reference lists. XeTeX writes no space characters into the PDF text
% layer, so some viewers (for example macOS Preview) join short words when text is copied
% ("Fosteringa", "AZAMAHM"). In the reference lists, every word boundary gets an invisible space
% glyph of zero net width, so copied text keeps its spaces; layout, line breaks and hyphenation are
% unchanged. The tokens are appended to the existing class transitions (ucharclasses font switching).
\makeatletter
% The glyph is added only when the next token is a real space, not at braces such as Pandoc's {[}.
\newcommand\bapilrealspace{\futurelet\bapil@next\bapil@realspace}
\newcommand\bapil@realspace{\ifx\bapil@next\@sptoken\ifhmode
    \count255=\spacefactor\relax
    \setbox\z@=\hbox{\XeTeXglyph\XeTeXcharglyph"20\relax}%
    \XeTeXglyph\XeTeXcharglyph"20\relax\kern-\wd\z@\relax\penalty\z@\relax
    \spacefactor=\count255\relax
  \fi\fi}
\newcommand\bapilrealspaces{\exhyphenpenalty=\@M % no line break at a hyphen in a reference (dates such as 2026-10-22)
  % web addresses in the reference lists break only at punctuation, preferably after "/", never inside a
  % word or a number; the lists are set ragged-right, so no line is stretched to fill the measure
  \def\UrlBreaks{\do\.\do\@\do\\\do\!\do\_\do\|\do\;\do\>\do\]\do\)\do\,\do\?\do\&\do\'\do+\do\=\do\#\do\-}%
  \def\UrlBigBreaks{\do\:\do\/}%
  \mathchardef\UrlBreakPenalty=300 \mathchardef\UrlBigBreakPenalty=100
  \raggedright
  \XeTeXinterchartokenstate=\@ne
  \count@=\z@
  \loop
    \XeTeXinterchartoks\count@\e@alloc@intercharclass@top=\expandafter{%
      \the\XeTeXinterchartoks\count@\e@alloc@intercharclass@top\bapilrealspace}%
    \advance\count@\@ne
  \ifnum\count@<\e@alloc@intercharclass@top\repeat}
\makeatother
